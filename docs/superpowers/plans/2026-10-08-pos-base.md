"""Compara las fallas actuales de pytest contra la línea base guardada.

Uso (desde backend/):
    python -m pytest -q -p no:cacheprovider -rfE > /tmp/pytest_out.txt
    python scripts/compare_baseline.py /tmp/pytest_out.txt ../docs/superpowers/plans/2026-10-08-pos-base-test-baseline.txt

Sale con código 1 si hay fallas o errores que no están en la línea base.
"""
import re
import sys


def failures(path):
    pattern = re.compile(r'^(FAILED|ERROR) (\S+)')
    result = set()
    with open(path, encoding='utf-8', errors='replace') as fh:
        for line in fh:
            match = pattern.match(line.strip())
            if match:
                result.add(f'{match.group(1)} {match.group(2).replace(chr(92), "/")}')
    return result


def main():
    current = failures(sys.argv[1])
    baseline = failures(sys.argv[2])
    new = sorted(current - baseline)
    fixed = sorted(baseline - current)
    print(f'Fallas actuales: {len(current)} | línea base: {len(baseline)}')
    if fixed:
        print(f'Ya no fallan ({len(fixed)}):')
        for item in fixed:
            print('  ', item)
    if new:
        print(f'FALLAS NUEVAS ({len(new)}):')
        for item in new:
            print('  ', item)
        sys.exit(1)
    print('Sin fallas nuevas.')


if __name__ == '__main__':
    main()
```

def module_required(module_name):
    """Exige acceso al módulo `module_name` (admin siempre pasa). Va debajo de @token_required."""
    def decorator(f):
        @wraps(f)
        def decorated_function(current_user, *args, **kwargs):
            if not current_user:
                logger.warning(
                    f"[SECURITY] Unauthorized access attempt | "
                    f"Path: {request.path} | Method: {request.method} | "
                    f"IP: {request.remote_addr} | Time: {datetime.utcnow().isoformat()}"
                )
                return jsonify({'error': 'Autenticación requerida'}), 401

            if not check_module_access(current_user, module_name):
                logger.warning(
                    f"[SECURITY] Forbidden access attempt | "
                    f"User: {current_user.email} | Role: {current_user.role} | "
                    f"Required module: {module_name} | "
                    f"Path: {request.path} | Method: {request.method} | "
                    f"IP: {request.remote_addr} | Time: {datetime.utcnow().isoformat()}"
                )
                return jsonify({
                    'error': 'Acceso denegado',
                    'message': f'No tienes acceso al módulo {module_name}'
                }), 403

            return f(current_user, *args, **kwargs)
        decorated_function._access_control = f'module:{module_name}'
        return decorated_function
    return decorator


def authenticated_only(f):
    """Marca un endpoint como accesible para cualquier usuario autenticado.

    No agrega validaciones: deja explícita la decisión para el test de cobertura de permisos.
    Va debajo de @token_required.
    """
    f._access_control = 'authenticated'
    return f
```

`functools.wraps` copia `__dict__` del wrapper interno al externo, así que `_access_control` llega hasta la view function que envuelve `@token_required`.

```python
"""Convierte los decoradores de acceso de backend/app/routes según el spec de pos-base.

Uso:
    python convert_access.py <routes_dir>                  # conversión completa de las rutas existentes
    python convert_access.py <routes_dir> <file> <Module>  # sólo un archivo (rutas portadas)
"""
import re
import sys
from pathlib import Path

MODULES = {
    'employees.py': 'Employees', 'job_positions.py': 'Employees', 'social_security.py': 'Employees',
    'schedules.py': 'Schedules', 'shifts.py': 'Schedules', 'schedule_summary.py': 'Schedules',
    'store_hours.py': 'Schedules', 'holidays.py': 'Schedules', 'vacation_periods.py': 'Schedules',
    'time_tracking.py': 'Schedules', 'csv_import.py': 'Schedules',
    'payroll.py': 'Payroll',
    'reports.py': 'Reports', 'ml_dashboard.py': 'Reports', 'ml_predictions.py': 'Reports',
    'expenses.py': 'Expenses', 'sales.py': 'Sales', 'menu.py': 'Menu',
}
LOCAL_ADMIN_FILES = {'holidays.py', 'payroll.py', 'social_security.py', 'time_tracking.py'}
ROUTE = re.compile(r'^\s*@\w+\.route\(')
LOCAL_DEF = re.compile(
    r"^def admin_required\(f\):\n"
    r"    @wraps\(f\)\n"
    r"    def decorated\(current_user, \*args, \*\*kwargs\):\n"
    r"        if (?:not current_user\.is_admin|current_user\.role != 'admin'):\n"
    r"            return jsonify\(\{'error': 'Se requieren permisos de administrador'\}\), 403\n"
    r"        return f\(current_user, \*args, \*\*kwargs\)\n"
    r"    return decorated\n\n?",
    re.M,
)
DECORATOR_IMPORT = re.compile(r'^from app\.utils\.decorators import (.+)$', re.M)
ACCESS_NAMES = ('admin_required', 'module_required', 'authenticated_only')


def convert(path, module):
    raw = path.read_bytes().decode('utf-8')
    crlf = '\r\n' in raw
    text = raw.replace('\r\n', '\n')

    if path.name in LOCAL_ADMIN_FILES:
        text, count = LOCAL_DEF.subn('', text)
        assert count == 1, f'{path.name}: esperaba 1 admin_required local, encontré {count}'

    lines = text.split('\n')
    out = []
    i = 0
    while i < len(lines):
        line = lines[i]
        out.append(line)
        if ROUTE.match(line):
            j = i + 1
            block = []
            while j < len(lines) and lines[j].strip().startswith('@'):
                block.append(lines[j])
                j += 1
            plain = [d.strip() for d in block if not ROUTE.match(d)]
            for d in block:
                if module and d.strip() == '@admin_required':
                    out.append(d.replace('@admin_required', f"@module_required('{module}')"))
                else:
                    out.append(d)
                if d.strip() == '@token_required' and plain == ['@token_required']:
                    out.append(d.replace('@token_required', '@authenticated_only'))
            i = j
            continue
        i += 1
    text = '\n'.join(out)

    used = {name for name in ACCESS_NAMES if re.search(rf'@{name}\b', text)}
    match = DECORATOR_IMPORT.search(text)
    if match:
        existing = [n.strip() for n in match.group(1).split(',')]
        others = [n for n in existing if n not in ACCESS_NAMES]
        names = others + [n for n in ACCESS_NAMES if n in used]
        text = text[:match.start()] + f"from app.utils.decorators import {', '.join(names)}" + text[match.end():]
    elif used:
        anchor = 'from app.utils.jwt_utils import token_required'
        assert anchor in text, f'{path.name}: no encontré el import de token_required'
        names = [n for n in ACCESS_NAMES if n in used]
        text = text.replace(anchor, anchor + f"\nfrom app.utils.decorators import {', '.join(names)}", 1)

    if crlf:
        text = text.replace('\n', '\r\n')
    path.write_bytes(text.encode('utf-8'))


def main():
    routes = Path(sys.argv[1])
    if len(sys.argv) == 4:
        convert(routes / sys.argv[2], sys.argv[3])
        return
    for path in sorted(routes.glob('*.py')):
        if path.name == '__init__.py':
            continue
        convert(path, MODULES.get(path.name))


if __name__ == '__main__':
    main()
```

@config_bp.route('/admin/config/branding', methods=['POST'])

@admin_required

    """Sube logo y/o fondo del banner al prefijo público `branding/`."""
    if request.content_length is not None and request.content_length > 2 * MAX_UPLOAD_BYTES + UPLOAD_OVERHEAD_BYTES:
        return jsonify({'error': 'Las imágenes superan los 10 MB'}), 413
    uploads = {field: request.files.get(field) for field in BRANDING_FIELDS}

    if not uploads:

    processed = {}

            processed[attribute] = process_image(upload.read(MAX_UPLOAD_BYTES + 1), max_side=max_side, key_prefix=prefix)

            return jsonify({'error': str(exc)}), 400

    storage = get_menu_storage()
    try:
        for body, key in processed.values():

        return jsonify({'error': 'No se pudo subir la imagen, probá de nuevo'}), 502

    if not config:

import { authService } from '@/services/authService'

const AuthContext = createContext(null)

const loadModules = async () => {
  try {
    return await permissionsService.getMyModules()
  } catch {
    return []
  }
}

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null)
  const [userModules, setUserModules] = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    initializeAuth()
  }, [])

  const initializeAuth = async () => {
    const token = authService.getToken()

    if (token) {
      try {
        const userData = await authService.getCurrentUser()
        setUser(userData)
        setUserModules(await loadModules())
      } catch (error) {
        setUser(null)
        setUserModules([])
      }
    }

    setLoading(false)
  }

  const login = async (email, password) => {
    const userData = await authService.login(email, password)
    setUser(userData)
    setUserModules(await loadModules())
    return userData
  }

  const logout = async () => {
    await authService.logout()
    setUser(null)
    setUserModules([])
  }

  const isAdmin = () => {
    return user?.role === 'admin'
  }

  // El admin tiene acceso a todo aunque falle la carga de módulos.
  const hasModuleAccess = (moduleName) => {
    if (isAdmin()) return true
    return userModules.some(module => module.name === moduleName)
  }

  return (
    <AuthContext.Provider value={{ user, userModules, loading, login, logout, isAdmin, hasModuleAccess }}>
      {children}
    </AuthContext.Provider>
  )
}

export const useAuth = () => {
  const context = useContext(AuthContext)
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider')
  }
  return context
}
