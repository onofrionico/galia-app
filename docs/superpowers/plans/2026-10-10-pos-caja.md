# POS de caja web (sub-proyecto 1b) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Interfaz web de caja del POS de Galia (plano de mesas, panel de venta, catálogo, cobro, mostrador y configuración) sobre la API del 1a, más los agregados de backend que la UI necesita: motivos de anulación, polling por cambios, mozos, compresión y gunicorn con threads.

**Architecture:** Todo el POS vive en `frontend/src/pos/`, con pantalla completa propia en `/pos` fuera del layout de Galia. El estado del servidor lo maneja React Query: cada acción es una mutation que guarda en la caché la venta completa que devuelve el backend, y un hook de polling (`/pos/changes` cada 3 s) invalida solo lo que cambió. La lógica pura (montos, permisos, invalidaciones, modificadores, plano) va en `pos/utils/` con tests de Vitest. El backend suma una migración `b1b1`, un servicio por tema y rutas delgadas, con el mismo estilo que el 1a.

**Tech Stack:** React 18, React Router 6, TanStack React Query 5, Tailwind (paleta Galia), lucide-react, axios, Vitest 1.6. Flask, Flask-SQLAlchemy, Alembic, pytest, Flask-Compress.

```python
"""Motivo de una anulación: uno del catálogo, texto libre o ambos."""
from app.extensions import db
from app.models.pos import CancelReason
from app.services.pos.common import parse_id, require_reason
from app.services.pos.errors import bad_request
from app.utils.validation import clean_str

MAX_REASON_LENGTH = 255


def resolve_reason(reason_id, reason):
    """Devuelve (texto, id) del motivo.

    Con `reason_id` (un motivo activo del catálogo) el texto es su nombre, más el detalle libre si viene.
    Sin `reason_id`, el texto libre es obligatorio.
    """
    if reason_id is None:
        return require_reason(reason), None
    found = db.session.get(CancelReason, parse_id(reason_id, 'El motivo'))
    if found is None or not found.is_active:
        raise bad_request('El motivo no existe')
    detail = clean_str(reason, MAX_REASON_LENGTH)
    text = f'{found.name} — {detail}' if detail else found.name
    return text[:MAX_REASON_LENGTH], found.id
```

```python
    item.cancel_reason = reason

def cancel_item(user, sale_id, item_id, reason, reason_id=None):
    reason, reason_id = resolve_reason(reason_id, reason)
    sale = load_sale(sale_id)
    require_status(sale, ACTIVE_SALE_STATUSES, 'Solo se anulan ítems de una venta abierta o en cobro')
    item = _find_item(sale, item_id)
    if item.status == 'pending':
        raise PosError('El ítem está pendiente: borralo en lugar de anularlo')
    if item.status == 'cancelled':
        raise PosError('El ítem ya está anulado')
    _cancel_item_row(item, user, reason, reason_id)
    stock_hooks.restore([item])
    recalc(sale)
    if sale.total < sale.paid_total:
        raise PosError('El total quedaría por debajo de lo ya pagado; anulá un pago primero')
    audit.record(sale, user, 'item_cancelled', item=item, product=item.product_name, quantity=item.quantity,
                 reason=reason, reason_id=reason_id)
    close_if_settled(sale, user)
    db.session.commit()
    return sale


def cancel_sale_rows(sale, user, reason, reason_id=None):
    """Anula una venta sin pagos activos: ítems, descuentos y stock. No hace commit."""
    confirmed = [i for i in sale.items if i.status == 'confirmed']
    for item in sale.items:
        if item.status != 'cancelled':
            _cancel_item_row(item, user, reason, reason_id)
    for discount in sale.discounts:
        if discount.cancelled_at is None:
            discount.cancelled_at = now()
            discount.cancelled_by = user.id
    stock_hooks.restore(confirmed)
    sale.status = 'cancelled'
    sale.cancelled_at = now()
    sale.cancelled_by = user.id
    sale.cancel_reason = reason
    sale.cancel_reason_id = reason_id
    recalc(sale)


def cancel_sale(user, sale_id, reason, reason_id=None):
    reason, reason_id = resolve_reason(reason_id, reason)
    sale = load_sale(sale_id)
    require_status(sale, tuple(ACTIVE_SALE_STATUSES) + ('closed',),
                   'Solo se anula una venta abierta, en cobro o cerrada sin pagos')
    if active_payments(sale):
        raise PosError('La venta tiene pagos; anulalos antes de anular la venta')
    if sale.status == 'closed':
        projection.unproject(sale)
        sale.closed_at = None
        sale.closed_by = None
    cancel_sale_rows(sale, user, reason, reason_id)
    audit.record(sale, user, 'cancelled', reason=reason, reason_id=reason_id)
    db.session.commit()
    return sale
```

```js
const whole = new Intl.NumberFormat('es-AR', {
  style: 'currency', currency: 'ARS', minimumFractionDigits: 0, maximumFractionDigits: 0,

const cents = new Intl.NumberFormat('es-AR', {
  style: 'currency', currency: 'ARS', minimumFractionDigits: 2, maximumFractionDigits: 2,
})

export const round2 = (value) => Math.round(value * 100) / 100

export function formatMoney(value) {
  const number = Number(value) || 0
  return Number.isInteger(round2(number)) ? whole.format(number) : cents.format(number)
}

// Texto tipeado por la caja: los puntos son de miles y la coma es decimal.
export function parseAmount(text) {
  const clean = String(text ?? '').replace(/\$/g, '').replace(/\s/g, '').replace(/\./g, '').replace(',', '.')
  if (clean === '') return NaN
  const number = Number(clean)
  return Number.isFinite(number) ? round2(number) : NaN
}

export function amountToText(value) {
  const number = round2(Number(value) || 0)
  return Number.isInteger(number) ? String(number) : number.toFixed(2).replace('.', ',')
}

export function formatQuantity(value) {
  return Number(value).toLocaleString('es-AR', { maximumFractionDigits: 3 })
}

export function discountLabel(discount) {
  return discount.kind === 'percent' ? `${Number(discount.value)}%` : formatMoney(discount.value)
```

```js
import { round2 } from './money'

const STEPS = [1000, 5000, 10000, 20000]

// Vista previa del vuelto; el vuelto real lo calcula el backend al registrar el pago.
export function changeFor(amount, tendered) {
  if (!Number.isFinite(amount) || !Number.isFinite(tendered) || tendered < amount) return null
  return round2(tendered - amount)
}

export function quickTenders(amount) {
  if (!(amount > 0)) return []
  const values = [round2(amount)]
  for (const step of STEPS) {
    const value = Math.ceil(amount / step) * step
    if (value > amount && !values.includes(value)) values.push(value)
  }
  return values.slice(0, 4)
}
```

```js
// Aplica una tecla del teclado numérico. `fresh` indica que el valor actual es una sugerencia
// (por ejemplo, el saldo) y la primera tecla lo reemplaza.
export function applyKey(text, key, fresh) {
  if (key === 'back') return fresh ? '' : text.slice(0, -1)
  if (key === ',') {
    if (fresh || text === '') return '0,'
    return text.includes(',') ? text : `${text},`
  }
  if (fresh || text === '0') return key
  const decimals = text.split(',')[1]
  if (decimals !== undefined && decimals.length >= 2) return text
  return text + key
}
```

```js
  if (!iso) return null

export function elapsedLabel(iso, now = new Date()) {
```

```js
export function posPermissions({ isAdmin = false, modules = [] } = {}) {
  const has = (name) => isAdmin || modules.includes(name)
  const pos = has('POS')
  const waiter = has('Camarero')
  return {
    pos,
    waiter,
    canUse: pos || waiter,
    charge: pos || has('Cobrar'),
    cancel: has('Anular'),
    discounts: has('Descuentos'),
  }
}

export function saleCounts(sale) {
  const items = sale?.items ?? []
  return {
    pending: items.filter((i) => i.status === 'pending').length,
    confirmed: items.filter((i) => i.status === 'confirmed').length,
  }
}

// Qué acciones muestra la UI para la venta. El backend valida igual.
export function saleActions(sale, perms) {
  const active = ACTIVE.includes(sale.status)
  const { pending, confirmed } = saleCounts(sale)
  const live = pending + confirmed
  return {
    addItems: sale.status === 'open',
    confirm: sale.status === 'open' && pending > 0,
    requestBill: sale.status === 'open' && sale.sale_type === 'salon' && live > 0,
    reopen: sale.status === 'billing',
    pay: perms.charge && active && live > 0 && sale.balance > 0,
    close: perms.charge && sale.status === 'billing' && sale.balance === 0,
    discount: active && confirmed > 0,
    move: active && sale.sale_type === 'salon',
    split: active && confirmed > 0 && sale.paid_total === 0,
    cancelSale: perms.cancel && active && sale.paid_total === 0,
    changeWaiter: perms.pos && active,
    history: perms.pos,
    cancelPayment: perms.cancel && ['billing', 'closed'].includes(sale.status),
  }
}

export function itemActions(item, sale, perms) {
  const active = ACTIVE.includes(sale.status)
  return {
    remove: item.status === 'pending' && sale.status === 'open',
    discount: item.status === 'confirmed' && active,
    cancel: perms.cancel && item.status === 'confirmed' && active,
  }
}
```

```js
import { round2 } from './money'

// Las mesas guardan posición y tamaño en unidades lógicas: el lienzo mide 100 de ancho.
export const LOGICAL_WIDTH = 100
export const MIN_CANVAS_HEIGHT = 60
export const GRID = 1
export const MIN_SIZE = 4

export function tableStatus(table) {
  const sales = table.sales ?? []
  if (sales.length === 0) return 'free'
  return sales.every((s) => s.status === 'billing') ? 'billing' : 'open'
}

export function tableSummary(table) {
  const sales = table.sales ?? []
  const openedAt = sales.map((s) => s.opened_at).filter(Boolean).sort()[0] ?? null
  return {
    count: sales.length,
    total: round2(sales.reduce((sum, s) => sum + Number(s.total || 0), 0)),
    waiter: sales[0]?.waiter?.name ?? null,
    openedAt,
  }
}

export const snap = (value) => Math.max(0, Math.round(value / GRID) * GRID)

export function clampPosition({ pos_x, pos_y, width }) {
  return { pos_x: Math.min(Math.max(0, pos_x), LOGICAL_WIDTH - width), pos_y: Math.max(0, pos_y) }
}

export function clampSize({ pos_x, width, height }) {
  return { width: Math.min(Math.max(MIN_SIZE, width), LOGICAL_WIDTH - pos_x), height: Math.max(MIN_SIZE, height) }
}

export function canvasHeight(tables) {
  return Math.max(MIN_CANVAS_HEIGHT, ...tables.map((t) => t.pos_y + t.height + 4))
}
```

```js
export function toggleOption(group, selected, optionId) {
  const inGroup = group.options.map((o) => o.id)
  const current = selected.filter((id) => inGroup.includes(id))
  if (current.includes(optionId)) return selected.filter((id) => id !== optionId)
  if (group.max_select === 1) return [...selected.filter((id) => !inGroup.includes(id)), optionId]
  if (current.length >= group.max_select) return selected
  return [...selected, optionId]
}

export function selectionError(groups, selected) {
  for (const group of groups) {
    const count = group.options.filter((o) => selected.includes(o.id)).length
    if (count < group.min_select) return `Elegí al menos ${group.min_select} en "${group.name}"`
  }
  return null
}

// Vista previa del importe del renglón; el importe real lo devuelve el backend al agregarlo.
export function linePreview(variant, groups, selected, quantity) {
  const deltas = groups
    .flatMap((g) => g.options)
    .filter((o) => selected.includes(o.id))
    .reduce((sum, o) => sum + Number(o.price_delta), 0)
  return round2((Number(variant.price) + deltas) * quantity)
}

export function groupHint(group) {
  if (group.min_select > 0) {
    const range = group.min_select === group.max_select ? group.min_select : `de ${group.min_select} a ${group.max_select}`
    return `elegí ${range}, obligatorio`
  }
  return `opcional, hasta ${group.max_select}`
}
```

```jsx
import { useEffect, useRef } from 'react'
import { X } from 'lucide-react'

// Pila de modales abiertos: Escape cierra solo el de arriba.
const stack = []
const WIDTHS = { sm: 'max-w-sm', md: 'max-w-lg', lg: 'max-w-3xl' }

export default function Modal({ title, onClose, children, footer, size = 'md' }) {
  const onCloseRef = useRef(onClose)
  onCloseRef.current = onClose

  useEffect(() => {
    const entry = {}
    stack.push(entry)
    const onKey = (event) => {
      if (event.key === 'Escape' && stack[stack.length - 1] === entry) onCloseRef.current()
    }
    window.addEventListener('keydown', onKey)
    return () => {
      stack.splice(stack.indexOf(entry), 1)
      window.removeEventListener('keydown', onKey)
    }
  }, [])

  return (
    <div
      className="fixed inset-0 z-40 flex items-center justify-center bg-black/40 p-4"
      onMouseDown={(event) => { if (event.target === event.currentTarget) onClose() }}
    >
      <div role="dialog" aria-modal="true" aria-label={title}
        className={`flex max-h-[90vh] w-full ${WIDTHS[size]} flex-col rounded-xl bg-white shadow-xl`}>
        <div className="flex items-center justify-between border-b px-5 py-3">
          <h2 className="text-lg font-semibold text-galia-plum">{title}</h2>
          <button type="button" onClick={onClose} aria-label="Cerrar" className="rounded-lg p-2 hover:bg-gray-100">
            <X size={20} />
          </button>
        </div>
        <div className="flex-1 overflow-y-auto px-5 py-4">{children}</div>
        {footer && <div className="border-t px-5 py-3">{footer}</div>}
      </div>
    </div>
  )
}
```

```jsx
import { createContext, useCallback, useContext, useMemo, useState } from 'react'

const ToastContext = createContext(null)
const STYLES = {
  error: 'bg-red-700 text-white',
  info: 'bg-galia-plum text-galia-cream',
  success: 'bg-green-700 text-white',
}

export function ToastProvider({ children }) {
  const [toasts, setToasts] = useState([])
  const dismiss = useCallback((id) => setToasts((list) => list.filter((t) => t.id !== id)), [])
  const show = useCallback((kind, message, action) => {
    const id = `${Date.now()}-${Math.random()}`
    setToasts((list) => [...list.slice(-3), { id, kind, message, action }])
    setTimeout(() => dismiss(id), action ? 8000 : 4000)
  }, [dismiss])
  const toast = useMemo(() => ({
    error: (message, action) => show('error', message, action),
    info: (message) => show('info', message),
    success: (message) => show('success', message),
  }), [show])

  return (
    <ToastContext.Provider value={toast}>
      {children}
      <div className="fixed bottom-4 left-1/2 z-50 flex w-[min(90vw,480px)] -translate-x-1/2 flex-col gap-2" aria-live="polite">
        {toasts.map((t) => (
          <div key={t.id} className={`flex items-center gap-3 rounded-lg px-4 py-3 text-sm shadow-lg ${STYLES[t.kind]}`}>
            <span className="flex-1">{t.message}</span>
            {t.action && (
              <button type="button" className="font-semibold underline"
                onClick={() => { dismiss(t.id); t.action.onClick() }}>
                {t.action.label}
              </button>
            )}
            <button type="button" aria-label="Cerrar aviso" onClick={() => dismiss(t.id)}>×</button>
          </div>
        ))}
      </div>
    </ToastContext.Provider>
  )
}

export function useToast() {
  const context = useContext(ToastContext)
  if (!context) throw new Error('useToast se usa dentro de ToastProvider')
  return context
}
```

```js
import { useCallback } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { useToast } from '../ui/Toasts'
import { errorMessage, isRetryable } from '../utils/errors'
import { posKeys } from './keys'

// Ejecuta una acción sobre una venta. La respuesta (la venta completa) reemplaza la caché.
// `run` devuelve la venta o null si falló (el error ya se mostró como aviso).
export function useSaleMutation() {
  const queryClient = useQueryClient()
  const toast = useToast()
  const { mutateAsync, isPending } = useMutation({
    mutationFn: ({ fn }) => fn(),
    onSuccess: (sale) => {
      if (sale?.id) {
        queryClient.setQueryData(posKeys.sale(sale.id), sale)
        queryClient.invalidateQueries({ queryKey: posKeys.events(sale.id) })
      }
      queryClient.invalidateQueries({ queryKey: posKeys.floors })
      queryClient.invalidateQueries({ queryKey: posKeys.sales })
    },
  })

  const run = useCallback(async (saleId, fn, { onError } = {}) => {
    try {
      return await mutateAsync({ fn })
    } catch (error) {
      if (onError?.(error)) return null
      const retry = isRetryable(error)
        ? { label: 'Reintentar', onClick: () => run(saleId, fn, { onError }) }
        : undefined
      toast.error(errorMessage(error), retry)
      if (saleId) queryClient.invalidateQueries({ queryKey: posKeys.sale(saleId) })
      return null
    }
  }, [mutateAsync, queryClient, toast])

  return { run, busy: isPending }
}
```

```js
import { useRef } from 'react'
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { posApi } from '../api/posApi'
import { keysToInvalidate } from '../utils/changes'
import { posKeys } from './keys'

const POLL_MS = 3000

// Consulta /pos/changes cada 3 s con la pestaña visible e invalida solo lo que cambió.
// Si falla la red, al reconectar refresca todo el POS.
export function usePosChanges() {
  const queryClient = useQueryClient()
  const lastId = useRef(null)
  const failed = useRef(false)

  useQuery({
    queryKey: ['pos-changes'],
    queryFn: async () => {
      try {
        const response = await posApi.getChanges(lastId.current)
        if (failed.current) {
          failed.current = false
          queryClient.invalidateQueries({ queryKey: posKeys.all })
        } else if (lastId.current !== null) {
          keysToInvalidate(response).forEach((queryKey) => queryClient.invalidateQueries({ queryKey }))
        }
        lastId.current = response.last_event_id
        return response.last_event_id
      } catch (error) {
        failed.current = true
        throw error
      }
    },
    refetchInterval: POLL_MS,
    refetchIntervalInBackground: false,
    retry: false,
    staleTime: 0,
    gcTime: 0,
  })
}
```

```js
import { useEffect, useState } from 'react'

export function useElementWidth(ref) {
  const [width, setWidth] = useState(0)
  useEffect(() => {
    const element = ref.current
    if (!element) return undefined
    const observer = new ResizeObserver(([entry]) => setWidth(entry.contentRect.width))
    observer.observe(element)
    setWidth(element.getBoundingClientRect().width)
    return () => observer.disconnect()
  }, [ref])
  return width
}
```

```js
// Hora actual que se actualiza cada `intervalMs` (para "abierta hace X min").
export function useNow(intervalMs = 30_000) {
  const [now, setNow] = useState(() => new Date())
  useEffect(() => {
    const id = setInterval(() => setNow(new Date()), intervalMs)
    return () => clearInterval(id)
  }, [intervalMs])
  return now
}
```

```jsx
    </Modal>

}
```

```jsx
export default function PaymentModal({ sale, onClose, onSaleClosed }) {
  const perms = usePosPermissions()
  const { data: methods = [] } = useLookup('payment-methods')
  const [methodId, setMethodId] = useState(null)
  const [amountText, setAmountText] = useState(amountToText(sale.balance))
  const [tenderedText, setTenderedText] = useState('')
  const [field, setField] = useState('amount')
  const [fresh, setFresh] = useState(true)
  const [error, setError] = useState('')

  const [cancelling, setCancelling] = useState(null)
  const requestId = useRef(newRequestId())
  const { run, busy } = useSaleMutation()

  const method = methods.find((m) => m.id === methodId) ?? methods[0] ?? null
  const isCash = method?.kind === 'cash'
  const amount = parseAmount(amountText)
  const tendered = parseAmount(tenderedText)
  const change = isCash ? changeFor(amount, tendered) : null
  const payments = sale.payments.filter((p) => p.is_active)

  const focusField = (name) => {
    setField(name)
    setFresh(true)
  }

  const onKey = (key) => {
    const setter = field === 'amount' ? setAmountText : setTenderedText
    setter((text) => applyKey(text, key, fresh))
    setFresh(false)
    setError('')
  }

  const resetForNext = (updated) => {
    setAmountText(amountToText(updated.balance))
    setTenderedText('')
    setField('amount')
    setFresh(true)

    if (!method) return setError('Elegí un medio de pago')
    if (!(amount > 0)) return setError('Indicá el monto')

    if (!result) return // el id se conserva: un reintento no duplica el pago
    requestId.current = newRequestId()
    if (result.status === 'closed') {
      const last = result.payments.filter((p) => p.is_active).at(-1)
      setClosedChange(last?.change ?? 0)
      return
    }
    resetForNext(result)
  }

            </>
          ) : (
            <p className="text-lg">Cobro completo.</p>
          )}
        </div>
      </Modal>
    )
  }

  return (

      <div className="grid gap-6 md:grid-cols-2">
        <div>
          <dl className="space-y-1 text-sm">

              <div key={payment.id} className="flex items-center justify-between text-gray-700">
                <dt>{payment.method_name}</dt>
                <dd className="flex items-center gap-1">
```

```jsx
export default function SalePanel({ saleId, siblings = [], initialModal = null, onSelect, onClose, onStartMove }) {
  const { data: sale, isLoading, isError } = useSale(saleId)
  const perms = usePosPermissions()
  const toast = useToast()
  const queryClient = useQueryClient()
  const { run, busy } = useSaleMutation()
  const [tab, setTab] = useState('ticket')
  const [modal, setModal] = useState(initialModal ? { type: initialModal } : null)

  if (isLoading) return <PanelMessage onClose={onClose}>Cargando venta…</PanelMessage>
  if (isError || !sale) return <PanelMessage onClose={onClose}>No se encontró la venta.</PanelMessage>

  const actions = saleActions(sale, perms)
  const { pending } = saleCounts(sale)
  const closeModal = () => setModal(null)

  // "Pedir cuenta" y "Cobrar" confirman antes la tanda pendiente.
  const confirmPendingFirst = async () => {
    if (!pending) return true
    const result = await run(sale.id, () => posApi.confirm(sale.id))
    if (result) toast.info('Se confirmó la tanda pendiente')
    return Boolean(result)
  }

  const handlers = {
    add: () => setModal({ type: 'add' }),
    confirm: () => run(sale.id, () => posApi.confirm(sale.id)),
    requestBill: async () => {
      if (await confirmPendingFirst()) await run(sale.id, () => posApi.requestBill(sale.id))
    },
    reopen: () => run(sale.id, () => posApi.reopen(sale.id)),
    pay: async () => {
      if (sale.sale_type === 'salon' && sale.status === 'open') {
        if (!(await confirmPendingFirst())) return
        if (!(await run(sale.id, () => posApi.requestBill(sale.id)))) return
      }
      setModal({ type: 'pay' })
    },
    close: async () => {
      if (await run(sale.id, () => posApi.close(sale.id))) onClose()
    },
    discount: () => setModal({ type: 'discount' }),
    move: onStartMove,
    split: () => setModal({ type: 'split' }),
    cancelSale: () => setModal({ type: 'cancel-sale' }),
    changeWaiter: () => setModal({ type: 'waiter' }),
  }

  const itemHandlers = {
    remove: (item) => run(sale.id, () => posApi.deleteItem(sale.id, item.id)),
    discount: (item) => setModal({ type: 'discount', item }),
    cancel: (item) => setModal({ type: 'cancel-item', item }),
    removeDiscount: (discount) => run(sale.id, () => posApi.cancelDiscount(discount.id)),
  }

  const tabClass = (active) =>
    `flex-1 py-2 text-sm font-medium ${active ? 'border-b-2 border-galia-plum text-galia-plum' : 'text-gray-600'}`

  return (
    <PanelShell>
      <SaleHeader sale={sale} onClose={onClose} canChangeWaiter={actions.changeWaiter} onChangeWaiter={handlers.changeWaiter} />
      {siblings.length > 1 && (
        <div className="flex gap-1 overflow-x-auto border-b px-3 py-2" role="tablist" aria-label="Cuentas de la mesa">
          {siblings.map((s, index) => (
            <button key={s.id} type="button" role="tab" aria-selected={s.id === sale.id} onClick={() => onSelect(s.id)}
              className={`whitespace-nowrap rounded-lg px-3 py-1.5 text-sm ${s.id === sale.id ? 'bg-galia-plum text-galia-cream' : 'bg-gray-100'}`}>
              Cuenta {index + 1} · #{s.number}
            </button>
          ))}
        </div>
      )}
      {actions.history && (
        <div className="flex border-b">
          <button type="button" className={tabClass(tab === 'ticket')} onClick={() => setTab('ticket')}>Ticket</button>
          <button type="button" className={tabClass(tab === 'history')} onClick={() => setTab('history')}>Historial</button>
        </div>
      )}
      <div className="min-h-0 flex-1 overflow-y-auto">
        {tab === 'history' && actions.history
          ? <SaleHistory saleId={sale.id} />
          : <Ticket sale={sale} perms={perms} handlers={itemHandlers} busy={busy} />}
      </div>
      <SaleFooter sale={sale} actions={actions} handlers={handlers} busy={busy} />

      {modal?.type === 'add' && <AddProductsModal sale={sale} onClose={closeModal} />}
      {modal?.type === 'pay' && (
        <PaymentModal sale={sale} onClose={closeModal} onSaleClosed={() => { closeModal(); onClose() }} />
      )}
      {modal?.type === 'discount' && <DiscountModal sale={sale} item={modal.item ?? null} onClose={closeModal} />}
      {modal?.type === 'split' && (
        <SplitModal sale={sale} onClose={closeModal} onDone={(newSale) => {
          closeModal()
          queryClient.invalidateQueries({ queryKey: posKeys.sale(sale.id) })
          onSelect(newSale.id)
        }} />
      )}
      {modal?.type === 'waiter' && <WaiterModal sale={sale} onClose={closeModal} />}
      {modal?.type === 'cancel-item' && (
        <CancelReasonModal title={`Anular ${modal.item.product_name}`} busy={busy} onClose={closeModal}
          onConfirm={async (body) => {
            if (await run(sale.id, () => posApi.cancelItem(sale.id, modal.item.id, body))) closeModal()
          }} />
      )}
      {modal?.type === 'cancel-sale' && (
        <CancelReasonModal title="Anular la venta" busy={busy} onClose={closeModal}
          onConfirm={async (body) => {
            if (await run(sale.id, () => posApi.cancelSale(sale.id, body))) {
              closeModal()
              onClose()
            }
          }} />
      )}
    </PanelShell>
  )
}
```

```jsx
import { useState } from 'react'
import { ArrowDown, ArrowUp, Pencil } from 'lucide-react'
import { posConfigApi } from '../api/posApi'
import { useConfigMutation } from '../hooks/useConfigMutation'
import { btn } from '../ui/buttons'

export default function SalonList({ salons, selectedId, onSelect }) {
  const [name, setName] = useState('')
  const create = useConfigMutation((body) => posConfigApi.createSalon(body))
  const update = useConfigMutation(({ id, ...body }) => posConfigApi.updateSalon(id, body))

  const submit = (event) => {
    event.preventDefault()
    if (!name.trim()) return
    create.mutate({ name: name.trim(), position: salons.length }, {
      onSuccess: (salon) => { setName(''); onSelect(salon.id) },
    })
  }

  // Reordena y guarda la posición de cada salón según su lugar en la lista.
  const move = (index, delta) => {
    const list = [...salons]
    const [salon] = list.splice(index, 1)
    list.splice(index + delta, 0, salon)
    list.forEach((s, position) => {
      if (s.position !== position) update.mutate({ id: s.id, position })
    })
  }

  const rename = (salon) => {
    const value = window.prompt('Nuevo nombre del salón', salon.name)
    if (value && value.trim() && value.trim() !== salon.name) update.mutate({ id: salon.id, name: value.trim() })
  }

  return (
    <aside className="flex w-72 flex-shrink-0 flex-col border-r border-gray-200 bg-white p-4">
      <h2 className="text-lg font-semibold text-galia-plum">Salones</h2>
      <ul className="mt-3 flex-1 space-y-1 overflow-y-auto">
        {salons.map((salon, index) => (
          <li key={salon.id} className={`flex items-center gap-1 rounded-lg px-2 py-1 ${salon.id === selectedId ? 'bg-galia-cream' : ''}`}>
            <button type="button" onClick={() => onSelect(salon.id)}
              className={`flex-1 truncate py-2 text-left ${salon.is_active ? '' : 'text-gray-400 line-through'}`}>
              {salon.name}
            </button>
            <button type="button" aria-label={`Subir ${salon.name}`} disabled={index === 0} onClick={() => move(index, -1)}
              className="rounded p-1 hover:bg-gray-100 disabled:opacity-30"><ArrowUp size={16} /></button>
            <button type="button" aria-label={`Bajar ${salon.name}`} disabled={index === salons.length - 1}
              onClick={() => move(index, 1)} className="rounded p-1 hover:bg-gray-100 disabled:opacity-30"><ArrowDown size={16} /></button>
            <button type="button" aria-label={`Renombrar ${salon.name}`} onClick={() => rename(salon)}
              className="rounded p-1 hover:bg-gray-100"><Pencil size={16} /></button>
          </li>
        ))}
      </ul>
      {selectedId && (
        <button type="button" className={btn('secondary', 'mb-3')}
          onClick={() => {
            const salon = salons.find((s) => s.id === selectedId)
            update.mutate({ id: salon.id, is_active: !salon.is_active })
          }}>
          {salons.find((s) => s.id === selectedId)?.is_active ? 'Desactivar salón' : 'Activar salón'}
        </button>
      )}
      <form onSubmit={submit} className="flex gap-2">
        <input value={name} onChange={(e) => setName(e.target.value)} placeholder="Nuevo salón" maxLength={100}
          aria-label="Nombre del salón nuevo" className="min-w-0 flex-1 rounded-lg border border-gray-300 px-3" />
```

```jsx
import { useRef, useState } from 'react'
import { useQuery, useQueryClient } from '@tanstack/react-query'
import { Plus } from 'lucide-react'
import { posConfigApi } from '../api/posApi'
import { posKeys } from '../hooks/keys'
import { useConfigMutation } from '../hooks/useConfigMutation'
import { useElementWidth } from '../hooks/useElementWidth'
import { btn } from '../ui/buttons'
import { LOGICAL_WIDTH, canvasHeight, clampPosition, clampSize, snap } from '../utils/floor'
import TableForm from './TableForm'

export default function FloorEditor({ salon }) {
  const queryClient = useQueryClient()
  const { data: tables = [] } = useQuery({
    queryKey: posKeys.tables(salon.id),
    queryFn: () => posConfigApi.listTables(salon.id),
  })
  const ref = useRef(null)
  const width = useElementWidth(ref)
  const scale = width / LOGICAL_WIDTH
  const [drafts, setDrafts] = useState({})
  const [selectedId, setSelectedId] = useState(null)
  const drag = useRef(null)
  const update = useConfigMutation(({ id, ...body }) => posConfigApi.updateTable(id, body))
  const create = useConfigMutation((body) => posConfigApi.createTable(body))

  const shown = tables.map((t) => ({ ...t, ...drafts[t.id] }))
  const selected = shown.find((t) => t.id === selectedId) ?? null

  const clearDraft = (id) => setDrafts(({ [id]: _removed, ...rest }) => rest)

  const save = (id, body, onDone) => update.mutate({ id, ...body }, {
    onSuccess: (table) => {
      queryClient.setQueryData(posKeys.tables(salon.id), (list = []) => list.map((t) => (t.id === table.id ? table : t)))
    },
    onSettled: onDone,
  })

  const startDrag = (event, table, mode) => {
    event.preventDefault()
    event.stopPropagation()
    event.currentTarget.setPointerCapture(event.pointerId)
    setSelectedId(table.id)
    drag.current = {
      id: table.id, mode, startX: event.clientX, startY: event.clientY,
      origin: { pos_x: table.pos_x, pos_y: table.pos_y, width: table.width, height: table.height },
    }
  }

  const onPointerMove = (event) => {
    const current = drag.current
    if (!current || !scale) return
    const dx = (event.clientX - current.startX) / scale
    const dy = (event.clientY - current.startY) / scale
    const { origin } = current
    const next = current.mode === 'move'
      ? clampPosition({ pos_x: snap(origin.pos_x + dx), pos_y: snap(origin.pos_y + dy), width: origin.width })
      : clampSize({ pos_x: origin.pos_x, width: snap(origin.width + dx), height: snap(origin.height + dy) })
    setDrafts((all) => ({ ...all, [current.id]: next }))
  }

  const onPointerUp = () => {
    const current = drag.current
    drag.current = null
    if (!current) return
    const draft = drafts[current.id]
    const changed = draft
      ? Object.fromEntries(Object.entries(draft).filter(([key, value]) => current.origin[key] !== value))
      : {}
    if (Object.keys(changed).length === 0) {
      clearDraft(current.id)
      return
    }
    save(current.id, changed, () => clearDraft(current.id))
  }

  const addTable = () => {
    const number = Math.max(0, ...tables.map((t) => t.number)) + 1
    create.mutate({ salon_id: salon.id, number, pos_x: 2, pos_y: 2, width: 10, height: 10 }, {
      onSuccess: (table) => setSelectedId(table.id),
    })
  }

  return (
    <div className="flex gap-4">
      <div className="min-w-0 flex-1">
        <div className="mb-3 flex items-center justify-between">
          <h2 className="text-lg font-semibold text-galia-plum">{salon.name}</h2>
          <button type="button" className={btn('primary')} onClick={addTable} disabled={create.isPending}>
            <Plus size={18} /> Agregar mesa
          </button>
        </div>
        <p className="mb-2 text-sm text-gray-600">Arrastrá las mesas para ubicarlas; la esquina inferior derecha cambia el tamaño.</p>
        <div ref={ref} onPointerMove={onPointerMove} onPointerUp={onPointerUp}
          className="relative w-full rounded-xl border border-dashed border-gray-300 bg-white"
          style={{
            height: width ? canvasHeight(shown) * scale : 400,
            backgroundImage: 'radial-gradient(circle, #d6d3d1 1px, transparent 1px)',
            backgroundSize: `${scale * 5}px ${scale * 5}px`,
          }}>
          {width > 0 && shown.map((table) => (
            <div key={table.id} onPointerDown={(e) => startDrag(e, table, 'move')}
              className={`absolute flex touch-none select-none items-center justify-center rounded-lg border-2 text-sm font-semibold ${
                table.id === selectedId ? 'border-galia-plum bg-galia-cream' : 'border-gray-400 bg-white'
              } ${table.is_active ? '' : 'opacity-40'}`}
              style={{ left: table.pos_x * scale, top: table.pos_y * scale, width: table.width * scale, height: table.height * scale, cursor: 'move' }}>
              {table.label}
              <span role="presentation" onPointerDown={(e) => startDrag(e, table, 'resize')}
                className="absolute bottom-0 right-0 h-4 w-4 cursor-se-resize rounded-tl bg-galia-plum/70" />
            </div>
          ))}
        </div>
      </div>
      <div className="w-72 flex-shrink-0">
        {selected
          ? <TableForm table={selected} busy={update.isPending} onSave={(body) => save(selected.id, body)} />
          : <p className="text-sm text-gray-600">Tocá una mesa para editar su número, nombre y capacidad.</p>}
      </div>
    </div>
  )
}
```

```jsx
            </span>
```

```jsx
              <button type="button" aria-label="Subir" disabled={index === 0} onClick={() => move(index, -1)}

            </span>
          </li>
        ))}
      </ul>
```

```jsx
import { useState } from 'react'
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { Plus } from 'lucide-react'
import { posConfigApi } from '@/pos/api/posApi'
import { errorMessage } from '@/pos/utils/errors'

// fields: [{ key, label, type: 'text' | 'number' | 'select' | 'checkbox', options?: [[value, label]] }]
const toValue = (field, value) => {
  if (field.type === 'number') return Number(value)
  if (field.type === 'checkbox') return Boolean(value)
  if (field.type === 'text') return String(value ?? '').trim()
  return value
}

const display = (field, value) => {
  if (field.type === 'checkbox') return value ? 'Sí' : 'No'
  if (field.type === 'select') return field.options.find(([v]) => v === value)?.[1] ?? value
  return value
}

export default function CatalogEditor({ resource, fields, emptyItem }) {
  const queryClient = useQueryClient()
  const key = ['pos-settings', resource]
  const { data: rows = [], isLoading, isError } = useQuery({ queryKey: key, queryFn: () => posConfigApi.listResource(resource) })
  const [editing, setEditing] = useState(null)
  const [error, setError] = useState('')

  const save = useMutation({
    mutationFn: (row) => {
      const body = Object.fromEntries(fields.map((f) => [f.key, toValue(f, row[f.key])]))
      return row.id ? posConfigApi.updateResource(resource, row.id, body) : posConfigApi.createResource(resource, body)
    },
    onSuccess: () => {
      setEditing(null)
      setError('')
      queryClient.invalidateQueries({ queryKey: key })
    },
    onError: (err) => setError(errorMessage(err)),
  })

  const input = (field) => {
    const value = editing[field.key]
    const set = (v) => setEditing({ ...editing, [field.key]: v })
    if (field.type === 'checkbox') {
      return <input type="checkbox" checked={Boolean(value)} onChange={(e) => set(e.target.checked)} />
    }
    if (field.type === 'select') {
      return (
        <select value={value} onChange={(e) => set(e.target.value)} className="w-full rounded border px-2 py-1.5">
          {field.options.map(([v, label]) => <option key={v} value={v}>{label}</option>)}
        </select>
      )
    }
    return (
      <input type={field.type === 'number' ? 'number' : 'text'} step="0.01" value={value ?? ''}
        onChange={(e) => set(e.target.value)} className="w-full rounded border px-2 py-1.5" />
    )
  }

  return (
    <div className="space-y-4">
      {editing ? (
        <form onSubmit={(e) => { e.preventDefault(); save.mutate(editing) }} className="space-y-3 rounded-lg bg-white p-4 shadow">
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
            {fields.map((field) => (
              <label key={field.key} className="text-sm">
                <span className="block text-gray-700">{field.label}</span>
                <span className="mt-1 block">{input(field)}</span>
              </label>
            ))}
          </div>
          {error && <p className="text-sm text-red-700">{error}</p>}
          <div className="flex gap-2">
            <button type="submit" disabled={save.isPending} className="rounded bg-galia-plum px-4 py-2 text-galia-cream disabled:opacity-50">Guardar</button>
            <button type="button" onClick={() => { setEditing(null); setError('') }} className="rounded border px-4 py-2">Cancelar</button>
          </div>
        </form>
      ) : (
        <button type="button" onClick={() => setEditing({ ...emptyItem })}
          className="flex items-center gap-2 rounded-lg bg-galia-plum px-4 py-2 text-galia-cream">
          <Plus size={18} /> Agregar
        </button>
      )}
      {isLoading && <p className="text-gray-500">Cargando…</p>}
      {isError && <p className="text-red-700">No se pudo cargar la lista.</p>}
      <div className="overflow-x-auto rounded-lg bg-white shadow">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left">
            <tr>{fields.map((f) => <th key={f.key} className="px-4 py-2">{f.label}</th>)}<th /></tr>
          </thead>
          <tbody>
            {rows.map((row) => (
              <tr key={row.id} className={`border-t ${row.is_active === false ? 'text-gray-400' : ''}`}>
                {fields.map((f) => <td key={f.key} className="px-4 py-2">{display(f, row[f.key])}</td>)}
                <td className="px-4 py-2 text-right">
                  <button type="button" onClick={() => setEditing({ ...row })} className="text-galia-plum underline">Editar</button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  )
}
```
